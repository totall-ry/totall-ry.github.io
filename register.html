import { doc, getDoc, setDoc } from "https://www.gstatic.com/firebasejs/11.9.1/firebase-firestore.js"; document.getElementById('registrationForm').addEventListener('submit', async function(e) { e.preventDefault(); const db = window.db; if (!db) { alert("Database connection initializing. Please try again."); return; } const usernameInput = document.getElementById('username').value.trim().toLowerCase(); const passwordInput = document.getElementById('password').value; const fullNameInput = document.getElementById('fullName').value.trim(); const universityIdInput = document.getElementById('universityId').value.trim(); const emailInput = document.getElementById('email').value.trim(); try { // 1. Check if the chosen username document already exists in your 'users' collection const userRef = doc(db, "users", usernameInput); const userSnap = await getDoc(userRef); if (userSnap.exists()) { // 2. Block the signup pipeline immediately if a match is found alert(`Registration Error: The username "${usernameInput}" is already taken. Please choose another one.`); return; // Stops the function execution dead in its tracks } // 3. Securely build the payload since the name is unique const userPayload = { password: passwordInput, role: 'student', fullName: fullNameInput, universityId: universityIdInput, preferredEmail: emailInput }; // 4. Save into Firestore database instance await setDoc(userRef, userPayload); alert('Account successfully registered! You can now log in.'); window.location.href = 'index.html'; } catch (err) { alert("Registration pipeline error: " + err.message); } });
